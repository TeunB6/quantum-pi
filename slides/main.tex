\begin{frame}{Taylor Series}
    The first step to more complicated formulas is using functions which result in multiples of $\pi$.\pause
    \\
    For example, $\arctan{1} = \dfrac{\pi}{4}$. \pause

    The taylor expansion for $\arctan x$ is $T(x) = \sum\limits_{k=0}^\infty\dfrac{(-1)^kx^{2k+1}}{2k+1}$.\pause

    Thus we find $\dfrac{\pi}{4} = \sum\limits_{k=0}^\infty \dfrac{(-1)^k}{2k+1}$. \pause

    \begin{highlight}[Note]
        This series is named the Leibniz formula for $\pi$
    \end{highlight}
\end{frame}

\begin{frame}{Quantum Bit}
    \begin{itemize}
        \item Basekets $\ket{0}$ and $\ket{1}$
        \item Generalized state $\ket{\psi} = c_0 \ket{0} + c_1\ket{1}$
        \item Due to normalization probability of finding the bit in state $\ket{k}$ is given by $|c_k|^2$
    \end{itemize}

    \begin{block}{Note}
        
        This actually generalizes, to qudits where we have basekets $\ket{0}, \ket{1}...\ket{d-1}$ and $\ket{\psi} = \sum\limits_{i=0}^{d-1}c_i\ket{i}$
    \end{block}
\end{frame}

\begin{frame}{Logic Gates}
\begin{block}{Hadamard Gate}
    \[
    \begin{minipage}{0.2\textwidth}
        \Qcircuit @C=1.0em @R=0.8em {
        & \gate{H} & \qw
        }
    \end{minipage}
    \]
    Sets any input qubit to an equal superposition between $\ket{0}$ and $\ket{1}$
    
\end{block}
\pause
\begin{block}{Pauli-Z Gate}
    \[
    \begin{minipage}{0.2\textwidth}
        \Qcircuit @C=1.0em @R=0.8em {
        & \gate{Z^\theta} & \qw
        }
    \end{minipage}
    \]

    This gate rotates around the Z direction of the Bloch Sphere, in matrix form $Z =\begin{bmatrix}
        1 & 0 \\
        0 & e^{i\pi\theta}
    \end{bmatrix}$. This adds a phase term to the quantum state.
\end{block}
\end{frame}

\begin{frame}{Logic Gates}
    \begin{block}{Pauli-X (NOT) Gate}
        \[
    \begin{minipage}{0.2\textwidth}
        \Qcircuit @C=1.0em @R=0.8em {
        & \gate{Z^\theta} & \qw
        }
    \end{minipage}
    \]

    This gate is the quantum equivalent of the classical NOT gate in matrix form $X =\begin{bmatrix}
        0 & 1 \\
        1 & 0
    \end{bmatrix}$. 
    \end{block}
    \pause

    \begin{block}{Controlled Gates}
     \[
    \begin{minipage}{0.2\textwidth}
        \Qcircuit @C=1.0em @R=0.8em {
        & \lstick{0} & \ctrl{1} & \qw\\
        & \lstick{1} &  \gate{X} & \qw
        }
    \end{minipage}
    \] 
    A controlled gate only affects the target qubit if the control bit is in state $\ket{1}$. This can thus be used to create entangled states.
    \end{block}
\end{frame}
